# Spike for ADR-0001 §6; see main.cpp.
cmake_minimum_required(VERSION 3.28)
project(tante_spike_native_arithmetic LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SPIKE_SANITIZE "Build with sanitizers" OFF)

find_path(FLINT_INCLUDE_DIR flint/flint.h REQUIRED)
find_path(GMP_INCLUDE_DIR gmp.h REQUIRED)
find_path(MPFR_INCLUDE_DIR mpfr.h REQUIRED)
find_library(FLINT_LIBRARY NAMES flint libflint REQUIRED)
find_library(MPFR_LIBRARY NAMES mpfr libmpfr REQUIRED)
find_library(GMP_LIBRARY NAMES gmp libgmp REQUIRED)

add_executable(spike main.cpp)
target_include_directories(spike SYSTEM PRIVATE ${FLINT_INCLUDE_DIR} ${MPFR_INCLUDE_DIR} ${GMP_INCLUDE_DIR})
target_link_libraries(spike PRIVATE ${FLINT_LIBRARY} ${MPFR_LIBRARY} ${GMP_LIBRARY})
if(NOT WIN32)
  find_package(Threads REQUIRED)
  target_link_libraries(spike PRIVATE Threads::Threads m)
endif()

if(MSVC)
  target_compile_options(spike PRIVATE /W4 /WX /permissive- /utf-8 /external:W0)
else()
  target_compile_options(spike PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror)
endif()

if(SPIKE_SANITIZE)
  if(MSVC)
    target_compile_options(spike PRIVATE /fsanitize=address /Zi)
    target_link_options(spike PRIVATE /INCREMENTAL:NO)
  else()
    target_compile_options(spike PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
    target_link_options(spike PRIVATE -fsanitize=address,undefined)
  endif()
endif()
